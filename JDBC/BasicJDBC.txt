import java.sql.*;

public class BasicJDBC {

    public static void main(String[] args) {

        try {

            // Load JDBC Driver
            Class.forName("com.mysql.cj.jdbc.Driver");

            // Establish Connection
            Connection con = DriverManager.getConnection(
                    "jdbc:mysql://localhost:3306/emp",
                    "root",
                    "root");

            // Create Statement
            Statement stmt = con.createStatement();

            // Create Table
            stmt.executeUpdate(
                    "CREATE TABLE IF NOT EXISTS Employee(" +
                    "id INT PRIMARY KEY, " +
                    "name VARCHAR(50), " +
                    "salary INT)");

            System.out.println("Table Created Successfully");

            // Insert Record
            stmt.executeUpdate(
                    "INSERT INTO Employee VALUES(1, 'Atharv', 50000)");

            stmt.executeUpdate(
                    "INSERT INTO Employee VALUES(2, 'Rahul', 40000)");

            System.out.println("Records Inserted Successfully");

            // Display Records
            System.out.println("\nEmployee Records:");

            ResultSet rs = stmt.executeQuery("SELECT * FROM Employee");

            while (rs.next()) {

                System.out.println(
                        rs.getInt("id") + "\t" +
                        rs.getString("name") + "\t" +
                        rs.getInt("salary"));
            }

            // Update Record
            stmt.executeUpdate(
                    "UPDATE Employee SET salary = 55000 WHERE id = 1");

            System.out.println("\nRecord Updated Successfully");

            // Delete Record
            stmt.executeUpdate(
                    "DELETE FROM Employee WHERE id = 2");

            System.out.println("Record Deleted Successfully");

            // Display Final Table
            System.out.println("\nFinal Employee Table:");

            rs = stmt.executeQuery("SELECT * FROM Employee");

            while (rs.next()) {

                System.out.println(
                        rs.getInt("id") + "\t" +
                        rs.getString("name") + "\t" +
                        rs.getInt("salary"));
            }

            // Close Connection
            con.close();

            System.out.println("\nConnection Closed");

        }

        catch (Exception e) {

            System.out.println("Error: " + e);
        }
    }
}
