// Chat System using Multithreading in Java


class ChatUser extends Thread {

    private String userName;
    private boolean running = true;
    private boolean suspended = false;

    // Constructor
    public ChatUser(String name) {
        this.userName = name;
    }

    // Send Message
    public void sendMessage(String msg) {
        System.out.println(userName + " sends: " + msg);
    }

    // Receive Message
    public void receiveMessage(String msg) {
        System.out.println(userName + " receives: " + msg);
    }

    // Suspend Thread
    public synchronized void suspendUser() {
        suspended = true;
    }

    // Resume Thread
    public synchronized void resumeUser() {
        suspended = false;
        notifyAll();
    }

    // Stop Thread
    public void stopUser() {
        running = false;
    }

    // Thread Execution
    public void run() {

        int count = 1;

        while (running && count <= 5) {

            try {

                synchronized (this) {

                    while (suspended) {
                        wait();
                    }
                }

                sendMessage("Hello " + count);

                receiveMessage("Reply " + count);

                Thread.sleep(1000);

                count++;

            }

            catch (InterruptedException e) {

                System.out.println(userName + " interrupted.");
            }
        }

        System.out.println(userName + " chat ended.");
    }
}

public class ChatSystem {

    public static void main(String[] args) {

        // Create Threads
        ChatUser user1 = new ChatUser("Alice");

        ChatUser user2 = new ChatUser("Bob");

        ChatUser user3 = new ChatUser("Admin");

        // Set Priorities
        user1.setPriority(Thread.NORM_PRIORITY);

        user2.setPriority(Thread.MIN_PRIORITY);

        user3.setPriority(Thread.MAX_PRIORITY);

        // Start Threads
        user1.start();

        user2.start();

        user3.start();

        // Check Status
        System.out.println("Is Alice alive? " + user1.isAlive());

        try {

            // Suspend Bob
            Thread.sleep(3000);

            System.out.println("\nSuspending Bob...\n");

            user2.suspendUser();

            // Resume Bob
            Thread.sleep(3000);

            System.out.println("\nResuming Bob...\n");

            user2.resumeUser();

            // Stop Alice
            Thread.sleep(3000);

            System.out.println("\nStopping Alice...\n");

            user1.stopUser();

            // Synchronization using join()
            user1.join();

            user2.join();

            user3.join();
        }

        catch (InterruptedException e) {

            System.out.println("Main thread interrupted.");
        }

        System.out.println("\nAll chats completed.");
    }
}
